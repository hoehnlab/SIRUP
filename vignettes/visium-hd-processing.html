Processing Visium HD BCR and TCR Data

This vignette walks through the Visium HD BCR/TCR processing workflow. It starts from RAD output, keeps spatial barcodes attached through the pRESTO/IgBLAST consensus workflow, produces AIRR-style tables with spatial coordinates, and plots BCR/TCR locations over the tissue.

For now, this vignette uses the full local thymus dataset from the walkthrough. We should replace this with a small packaged toy dataset later so the vignette can be run anywhere without large external files.

Create or Load the Seurat Object

If you need to create the Visium HD Seurat object directly from Space Ranger output, use Load10X_Spatial(). Visium HD loading requires arrow, which can be difficult to install, but SIRAP’s downstream plotting and analyses do not otherwise require arrow. We also find it easier to create a .csv version of the barcode mappings rather than relying on the read_parquet() function in arrow to read the parquet files directly.

# Build seurat object
m5_path <- "./analysis/M5_260129/outs/"
thy_m5 <- Load10X_Spatial(
  data.dir = m5_path,
  bin.size = c(8),
  image.name = "tissue_hires_image.png"
)

saveRDS(thy_m5, file.path(thymusDir, "data", "Thymus_VisiumHD_M5_8um_hires.rds"))

# Resave tissue locations as .csvs
m5_barcode_mapping <- read_parquet(file.path(m5_path, "barcode_mappings.parquet"))
write.csv(m5_barcode_mapping, file.path(thymusDir, "M5_barcode_mappings.csv"),

Load Spatial Metadata

The receptor reads carry the 2 um barcode in the sequence identifier. After pRESTO and IgBLAST, we join through the Visium HD barcode map to recover the 8 um and 16 um barcodes used for plotting and analysis.

barcodeMap <- readBarcodeMapping(barcodeMapPath)

positions8um <- readBarcodeMapping(positionsPath8um) %>%
  mutate(
    barcode_8um = barcode,
    x = pxl_col_in_fullres,
    y = max(pxl_row_in_fullres) - pxl_row_in_fullres
  )

Start from Raw IgBLAST Output

Extract the 2 um spatial barcode before pRESTO and do a first length/locus filter. filterSpatialAirr() keeps preprocessing in memory; we explicitly write AIRR tables and FASTA files only when we need pRESTO inputs.

bcrRaw <- data.table::fread(rawBCRIgblastPath)
tcrRaw <- data.table::fread(rawTCRIgblastPath)

bcrRaw <- extractSpatialBarcodes(
  bcrRaw,
  sequence_id_col = "sequence_id",
  barcode_col = "barcode_2um",
  regex = "[0-9]+_[0-9]+$",
  prefix = "s_002um_",
  suffix = "-1"
)

tcrRaw <- extractSpatialBarcodes(
  tcrRaw,
  sequence_id_col = "sequence_id",
  barcode_col = "barcode_2um",
  regex = "[0-9]+_[0-9]+$",
  prefix = "s_002um_",
  suffix = "-1"
)

bcrHeavy <- filterSpatialAirr(
  bcrRaw,
  loci = "IGH",
  min_v_length = 100,
  min_j_length = 10,
  min_junction_length = 1,
  barcode_col = "barcode_2um",
  add_full_id = TRUE
)

tcrBeta <- filterSpatialAirr(
  tcrRaw,
  loci = "TRB",
  min_v_length = 100,
  min_j_length = 10,
  min_junction_length = 1,
  barcode_col = "barcode_2um",
  add_full_id = TRUE
)

spatialReceptorSummary(bcrHeavy, barcode_col = "barcode_2um")
spatialReceptorSummary(tcrBeta, barcode_col = "barcode_2um")
bcrHeavySpatialTsv <- file.path(prestoInputDir, sprintf("%s_igblast_bcr_pass_length_spatial_barcode.tsv", sampleLower))
bcrHeavySpatialFasta <- file.path(prestoInputDir, sprintf("%s_igblast_bcr_pass_length_spatial_barcode.fasta", sampleLower))
tcrBetaSpatialTsv <- file.path(prestoInputDir, sprintf("%s_igblast_tcr_pass_length_spatial_barcode.tsv", sampleLower))

Final AIRR Tables with Spatial Coordinates

After pRESTO and IgBLAST, read the consensus db-pass files, extract the barcode from the pRESTO sequence identifier, remap to 8 um/16 um barcodes, add tissue positions, and collapse obvious spatial bleedover.

bcrConsensus <- airr::read_rearrangement(bcrPrestoPaths$db_pass)
tcrConsensus <- airr::read_rearrangement(tcrPrestoPaths$db_pass)

bcrConsensus <- extractSpatialBarcodes(
  bcrConsensus,
  sequence_id_col = "sequence_id",
  barcode_col = "barcode_2um",
  regex = ",([^,]+)"
)

tcrConsensus <- extractSpatialBarcodes(
  tcrConsensus,
  sequence_id_col = "sequence_id",
  barcode_col = "barcode_2um",
  regex = ",([^,]+)"
)

bcrConsensusFiltered <- filterSpatialAirr(
  bcrConsensus,
  loci = "IGH",
  min_v_length = 250,
  min_j_length = 30,
  min_junction_length = 10,
  productive_only = TRUE,
  min_consensus_count = 2,
  max_opening_gap = 2,
  barcode_col = "barcode_2um"
)

tcrConsensusFiltered <- filterSpatialAirr(
  tcrConsensus,
  loci = "TRB",
  min_v_length = 250,
  min_j_length = 30,
  min_junction_length = 10,
  productive_only = TRUE,
  min_consensus_count = 2,
  max_opening_gap = 2,
  barcode_col = "barcode_2um"
)

bcrConsensusFiltered <- bcrConsensusFiltered %>%
  mutate(barcode_2um_swapped = sub("^(.*_)(\\d+)_(\\d+)(-\\d+)$", "\\1\\3_\\2\\4", barcode_2um)) %>%
  left_join(
    barcodeMap %>% select(square_002um, square_008um, square_016um),
    by = c("barcode_2um_swapped" = "square_002um")
  ) %>%
  rename(barcode_8um = square_008um, barcode_16um = square_016um)

tcrConsensusFiltered <- tcrConsensusFiltered %>%
  mutate(barcode_2um_swapped = sub("^(.*_)(\\d+)_(\\d+)(-\\d+)$", "\\1\\3_\\2\\4", barcode_2um)) %>%
  left_join(
    barcodeMap %>% select(square_002um, square_008um, square_016um),
    by = c("barcode_2um_swapped" = "square_002um")
  ) %>%
  rename(barcode_8um = square_008um, barcode_16um = square_016um)

bcrConsensusFilteredPos <- addTissuePositions(
  bcrConsensusFiltered,
  positions8um,
  barcode_col = "barcode_8um",
  position_barcode_col = "barcode_8um",
  x_col = "x",
  y_col = "y"
) %>% filter(!is.na(x), !is.na(y))

tcrConsensusFilteredPos <- addTissuePositions(
  tcrConsensusFiltered,
  positions8um,
  barcode_col = "barcode_8um",
  position_barcode_col = "barcode_8um",
  x_col = "x",
  y_col = "y"
) %>% filter(!is.na(x), !is.na(y))

bcrCollapsed <- collapseIdenticalNeighbors(
  data = bcrConsensusFilteredPos,
  tissue_positions = positions8um,
  barcode = "barcode_8um",
  x = "x",
  y = "y",
  sequence = "sequence_alignment",
  consensus_count = "consensus_count",
  num_neighbors_non_boundary = 8L
)

tcrCollapsed <- collapseIdenticalNeighbors(
  data = tcrConsensusFilteredPos,
  tissue_positions = positions8um,
  barcode = "barcode_8um",
  x = "x",
  y = "y",
  sequence = "sequence_alignment",
  consensus_count = "consensus_count",
  num_neighbors_non_boundary = 8L
)

bcrTopBarcode <- collapseTopConsensusPerBarcode(bcrCollapsed, barcode_col = "barcode_2um", locus_col = "locus", ties = "all")
tcrTopBarcode <- collapseTopConsensusPerBarcode(tcrCollapsed, barcode_col = "barcode_2um", locus_col = "locus", ties = "all")

airr::write_rearrangement(bcrTopBarcode, finalBCRPath)
airr::write_rearrangement(tcrTopBarcode, finalTCRPath)

Create SIRAP Objects

Once the receptor table and spatial Seurat object are available, keep their relationship in a SIRAP object. The receptor data are stored as a tibble, and the resolved barcode column is recorded so downstream functions do not need to guess it repeatedly. Receptor rows that do not match the spatial object are retained; this is useful when a table also contains bulk or other nonspatial records.

bcrSirap <- createSIRAP(
  spatial = thymus,
  receptors = bcrTopBarcode,
  barcode_col = "barcode_8um",
  metadata = list(sample_id = sampleId, receptor_type = "BCR")
)

tcrSirap <- createSIRAP(
  spatial = thymus,
  receptors = tcrTopBarcode,
  barcode_col = "barcode_8um",
  metadata = list(sample_id = sampleId, receptor_type = "TCR")
)

bcrSirap
summary(tcrSirap)
spatialReceptorSummary(bcrSirap)

Plot BCRs and TCRs Over Tissue

plotReceptorExpression() and spatialDiscretePlot() draw points over the tissue image rather than over a blank coordinate plane. When passed a SIRAP object, plotReceptorExpression() obtains the Seurat object, receptor tibble, and barcode field from the container.

allBCRPlot <- plotReceptorExpression(
  object = bcrSirap,
  receptor_color = "white",
  receptor_size = 0.5,
  title = paste(sampleId, "all BCR-positive spots")
)

allTCRPlot <- plotReceptorExpression(
  object = tcrSirap,
  receptor_color = "#d95f02",
  receptor_size = 0.5,
  title = paste(sampleId, "all TCR-positive spots")
)

bcr_spots <- unique(bcrTopBarcode$barcode_8um)
tcr_spots <- unique(tcrTopBarcode$barcode_8um)
thymus[["sirap_receptor_type"]] <- dplyr::case_when(
  colnames(thymus) %in% bcr_spots & colnames(thymus) %in% tcr_spots ~ "BCR + TCR",
  colnames(thymus) %in% bcr_spots ~ "BCR",
  colnames(thymus) %in% tcr_spots ~ "TCR",
  TRUE ~ "0"
)

bcrTCROverlay <- spatialDiscretePlot(
  thymus,
  meta_col = "sirap_receptor_type",
  zero_value = "0",
  image.scale = "hires",
  point_size = 0.5,
  cols = c("BCR" = "#2c7fb8", "TCR" = "#d95f02", "BCR + TCR" = "white")
) +
  ggtitle(paste(sampleId, "BCR and TCR locations"))

cowplot::plot_grid(allBCRPlot, allTCRPlot, bcrTCROverlay, ncol = 3)
