Spatial Clones and BCR Trees

This vignette shows how to call clones, visualize BCR/TCR clone locations over tissue, build BCR lineage trees with dowser, and pair each tree with the spatial locations of its colored tips.

For now, this vignette uses the full local thymus dataset from the walkthrough. We should replace this with a small packaged toy dataset later.

Setup

library(SIRAP)
library(dplyr)
library(ggplot2)
library(cowplot)
library(airr)
library(Seurat)
library(SeuratObject)
library(scoper)
library(dowser)
library(alakazam)

sampleId <- "M5"
sampleLower <- tolower(sampleId)
nproc <- 8

thymusDir <- normalizePath("../../Thymus_VisiumHD", mustWork = TRUE)
outDir <- file.path(thymusDir, "SIRAP_walkthrough", sampleId)
finalAirrDir <- file.path(outDir, "03_final_spatial_airr")
plotDir <- file.path(outDir, "plots")
dir.create(plotDir, recursive = TRUE, showWarnings = FALSE)

seuratPath <- file.path(thymusDir, "data", sprintf("Thymus_VisiumHD_%s_8um_hires.rds", sampleId))
finalBCRPath <- file.path(finalAirrDir, sprintf("%s_bcrs_heavy_final_spatial_airr.tsv", sampleLower))
finalTCRPath <- file.path(finalAirrDir, sprintf("%s_tcrs_beta_final_spatial_airr.tsv", sampleLower))

bcrClonesPath <- file.path(outDir, sprintf("%s_bcr_hierarchical_clones.rds", sampleLower))
tcrClonesPath <- file.path(outDir, sprintf("%s_tcr_hierarchical_clones.rds", sampleLower))
bcrTreesPath <- file.path(outDir, sprintf("%s_bcr_pml_trees.rds", sampleLower))

thymus <- readRDS(seuratPath)
bcrTopBarcode <- airr::read_rearrangement(finalBCRPath)
tcrTopBarcode <- airr::read_rearrangement(finalTCRPath)

Call BCR and TCR Clones

SIRAP does not wrap scoper clone calling. Use scoper directly, then pass the resulting data frames to SIRAP plotting functions.

bcrClones <- scoper::hierarchicalClones(
  filter(bcrTopBarcode, productive),
  threshold = 0.1,
  cell_id = "sequence_id",
  summarize_clones = FALSE,
  nproc = nproc
)

tcrClones <- scoper::identicalClones(
  filter(tcrTopBarcode, productive),
  cell_id = "sequence_id",
  summarize_clones = FALSE,
  nproc = nproc
)

saveRDS(bcrClones, bcrClonesPath)
saveRDS(tcrClones, tcrClonesPath)

If clone calling has already been run, load the saved RDS files instead.

bcrClones <- readRDS(bcrClonesPath)
tcrClones <- readRDS(tcrClonesPath)

Keep each clone table paired with its spatial transcriptomics data in a SIRAP object. createSIRAP() converts the receptor table to a tibble and records the receptor field mappings in the SIRAP metadata without copying receptor columns into Seurat metadata.

bcrSirap <- createSIRAP(
  spatial = thymus,
  receptors = bcrClones,
  barcode_col = "barcode_8um",
  clone_col = "clone_id",
  metadata = list(sample_id = sampleId, receptor_type = "BCR")
)

tcrSirap <- createSIRAP(
  spatial = thymus,
  receptors = tcrClones,
  barcode_col = "barcode_8um",
  clone_col = "clone_id",
  metadata = list(sample_id = sampleId, receptor_type = "TCR")
)

Choose Clones to Plot

For a first pass, choose a couple of clones with enough tips to be interesting but not so many that the tree becomes hard to read.

bcrCloneCounts <- bcrClones %>%
  filter(!is.na(clone_id)) %>%
  count(clone_id, sort = TRUE)

tcrCloneCounts <- tcrClones %>%
  filter(!is.na(clone_id)) %>%
  count(clone_id, sort = TRUE)

bcrCloneIds <- bcrCloneCounts %>%
  filter(n >= 30, n <= 50) %>%
  slice_head(n = 2) %>%
  pull(clone_id)

tcrCloneIds <- tcrCloneCounts %>%
  filter(n >= 30, n <= 50) %>%
  slice_head(n = 2) %>%
  pull(clone_id)

bcrCloneCounts %>% filter(clone_id %in% bcrCloneIds)
tcrCloneCounts %>% filter(clone_id %in% tcrCloneIds)

Plot Clone Locations Over Tissue

plotReceptors() uses the stored receptor tibble and receptor field metadata directly. The values argument selects the clones to draw.

bcrClonePlot <- plotReceptors(
  object = bcrSirap,
  group_col = "clone_id",
  values = bcrCloneIds,
  receptor_size = 1.7,
  title = "Selected BCR clones"
)

tcrClonePlot <- plotReceptors(
  object = tcrSirap,
  group_col = "clone_id",
  values = tcrCloneIds,
  receptor_size = 1.7,
  title = "Selected TCR clones"
)

cowplot::plot_grid(bcrClonePlot, tcrClonePlot, ncol = 2)

Build BCR Trees with Dowser

SIRAP keeps tree building in dowser, then focuses on coloring and pairing trees with spatial plots.

germlineDir <- path.expand("~/software/germlines/imgt/human/vdj")
refs <- dowser::readIMGT(germlineDir)

bcrTreeInput <- bcrClones %>%
  filter(clone_id %in% bcrCloneIds, !is.na(clone_id))

bcrTreeInput <- dowser::createGermlines(bcrTreeInput, refs, id = "sequence_id")

bcrFormattedClones <- dowser::formatClones(
  bcrTreeInput,
  id = "sequence_id",
  germ = "germline_alignment",
  traits = c("barcode_8um"),
  columns = c("x", "y", "barcode_8um"),
  filterstop = FALSE,
  collapse = TRUE,
  add_count = "consensus_count" %in% colnames(bcrTreeInput),
  num_fields = if ("consensus_count" %in% colnames(bcrTreeInput)) "consensus_count" else NULL
)

bcrTrees <- dowser::getTrees(
  bcrFormattedClones,
  method = "pml",
  nproc = nproc
)

saveRDS(bcrTrees, bcrTreesPath)

If trees have already been built, load them from disk.

bcrTrees <- readRDS(bcrTreesPath)

Pair BCR Trees with Spatial Tip Locations

plotSpatialCloneTreePairs() assigns tip colors, draws the BCR tree with ggtree, and creates a matching spatial plot over the tissue. By default, each tip gets its own color.

pairedTreeSpatialPlots <- plotSpatialCloneTreePairs(
  bcrTrees,
  object = bcrSirap,
  clone_ids = bcrCloneIds,
  id_col = "sequence_id",
  treescale = 0.01,
  zoom = FALSE
)

firstCloneId <- names(pairedTreeSpatialPlots)[1]
cowplot::plot_grid(
  pairedTreeSpatialPlots[[firstCloneId]]$tree,
  pairedTreeSpatialPlots[[firstCloneId]]$spatial_clone,
  ncol = 2,
  rel_widths = c(1.2, 1)
)

For larger clones, coloring every tip can be hard to read. clusterTreesTips() cuts the k - 1 longest branches in each tree, creating k connected subtrees. Passing that new column to plotSpatialCloneTreePairs() colors the tree and spatial plot by genetically similar tip groups.

bcrTreesClustered <- clusterTreesTips(
  bcrTrees,
  k = 6,
  id_col = "sequence_id",
  cluster_col = "tree_cluster"
)

pairedTreeSpatialClusterPlots <- plotSpatialCloneTreePairs(
  bcrTreesClustered,
  object = bcrSirap,
  clone_ids = bcrCloneIds,
  id_col = "sequence_id",
  color_by = "tree_cluster",
  treescale = 0.01,
  zoom = FALSE
)

firstCloneId <- names(pairedTreeSpatialClusterPlots)[1]
cowplot::plot_grid(
  pairedTreeSpatialClusterPlots[[firstCloneId]]$tree,
  pairedTreeSpatialClusterPlots[[firstCloneId]]$spatial_clone,
  ncol = 2,
  rel_widths = c(1.2, 1)
)

If you want SIRAP to choose a separate cluster number for each tree, set k = "auto" and adjust k_range or improvement_threshold as needed.

bcrTreesAutoClustered <- clusterTreesTips(
  bcrTrees,
  k = "auto",
  id_col = "sequence_id",
  cluster_col = "tree_cluster",
  k_range = 1:8,
  improvement_threshold = 0.10
)

bcrTreesAutoClustered$tree_cluster_k
attr(bcrTreesAutoClustered, "tree_cluster_scores")

Export all paired tree/spatial plots to a single PDF.

spatialCloneTreePairsToPDF(
  pairedTreeSpatialPlots,
  file.path(plotDir, sprintf("%s_bcr_tree_spatial_pairs.pdf", sampleLower)),
  width = 12,
  height = 6
)
